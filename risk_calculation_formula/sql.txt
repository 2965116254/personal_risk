'''
按工作负责人及违章类别统计次数
'''
SELECT 
    w.work_principal_uid,
    LEFT(p.peccancy_code, 1) AS first_letter,          -- 违章代码首字母
    COUNT(*) AS peccancy_count
FROM sp_pd_wticket_base w
INNER JOIN sp_ss_uq_peccancy_list_log p 
    ON w.work_principal_uid = p.work_principal_uid    -- 请根据实际外键字段名调整
WHERE YEAR(p.record_date) = 2025                       -- 2025年违章记录
  AND w.work_principal_uid IS NOT NULL                 -- 排除空负责人
GROUP BY w.work_principal_uid, first_letter
ORDER BY w.work_principal_uid, peccancy_count DESC;

        id AS ticket_id,

)

WHERE YEAR(p.record_date) = 2025
GROUP BY sm.member_uid, first_letter
ORDER BY sm.member_uid, peccancy_count DESC;

全部工作票的作业总人数（简单求和）

SELECT 
    SUM(CAST(work_member_count AS SIGNED)) AS total_worker_count
FROM sp_pd_wticket_base
WHERE work_member_count IS NOT NULL 
  AND work_member_count REGEXP '^[0-9]+$';   -- 过滤非数字脏数据

  按年度/月度统计作业总人数（示例：2025年）

  SELECT 
    YEAR(create_time) AS year,
    MONTH(create_time) AS month,
    SUM(CAST(work_member_count AS SIGNED)) AS total_worker_count
FROM sp_pd_wticket_base
WHERE create_time >= '2025-01-01' AND create_time < '2026-01-01'
  AND work_member_count IS NOT NULL
GROUP BY YEAR(create_time), MONTH(create_time)
ORDER BY year, month;

‘’‘
负责人的人员性质，查询作业计划表sp_ss_rc_work_plan，获取作业主体task_main，作业计划编号work_code，查询sp_ss_uq_operator_archive，获取单位属性
unit_attribute，对比作业主体和单位属性，如果一致则输出人员性质，不一致则提醒用户人员性质不一致
’‘’

SELECT 
    wp.work_code AS 作业计划编号,
    wp.task_main AS 作业主体,
    wp.work_master_uname AS 工作负责人,
    oa.unit_attribute AS 单位属性,
    CASE 
        WHEN oa.user_uid IS NULL THEN '无法对比：未找到该负责人的档案记录'
        WHEN CAST(wp.task_main AS UNSIGNED) = 1 AND CAST(oa.unit_attribute AS UNSIGNED) = 1 
            THEN '一致：本单位-系统内人员'
        WHEN CAST(wp.task_main AS UNSIGNED) = 2 AND CAST(oa.unit_attribute AS UNSIGNED) IN (2, 21) 
            THEN '一致：外单位-系统外人员'
        WHEN wp.task_main IS NULL OR oa.unit_attribute IS NULL 
            THEN '无法对比（缺少作业主体或人员档案）'
        ELSE '不一致：人员性质与作业主体不匹配'
    END AS 对比结果
FROM 
    sp_ss_rc_work_plan wp
LEFT JOIN 
    sp_ss_uq_operator_archive oa 
    ON wp.work_master_uid = oa.user_uid
    -- 如果仍需过滤启用状态，取消下一行注释
    -- AND oa.archive_state = '1'
WHERE 
    wp.work_master_uid IS NOT NULL;

'''
工作负责人(含小组工作负责人)组织同类型作业次数，查询操作票表sp_pd_wticket_base，确认负责人，查询作业计划表sp_ss_rc_work_plan，获取作业类型task_type
，再查负责人在查询作业计划表sp_ss_rc_work_plan中的所有符合那条工作计划的作业类型次数
'''

WITH task_stats AS (
    SELECT
        work_master_uid AS leader_uid,
        work_master_uname AS leader_name,
        task_type,
        COUNT(1) AS total_count,
        -- 统计1年内、2年内次数（需要时间字段 create_time）
        SUM(CASE WHEN create_time >= DATE_SUB(CURDATE(), INTERVAL 1 YEAR) THEN 1 ELSE 0 END) AS cnt_last_1year,
        SUM(CASE WHEN create_time >= DATE_SUB(CURDATE(), INTERVAL 2 YEAR) THEN 1 ELSE 0 END) AS cnt_last_2year
    FROM sp_ss_rc_work_plan
    WHERE work_master_uid IS NOT NULL
      AND task_type IS NOT NULL
    GROUP BY work_master_uid, work_master_uname, task_type
)

SELECT
    leader_uid,
    leader_name,
    task_type,
    -- 作业类型名称
    CASE ROUND(task_type, 0)
        WHEN 1 THEN '预试'
        WHEN 2 THEN '定检'
        WHEN 3 THEN '巡视'
        WHEN 4 THEN '检修'
        WHEN 5 THEN '维护'
        WHEN 6 THEN '消缺'
        WHEN 7 THEN '操作'
        WHEN 8 THEN '验收'
        WHEN 9 THEN '安装、更换作业'
        WHEN 10 THEN '测量'
        WHEN 11 THEN '带电'
        WHEN 12 THEN '工器具试验'
        WHEN 13 THEN '配合(生产)'
        WHEN 14 THEN '其他(生产)'
        WHEN 15 THEN '反措'
        ELSE '未知类型'
    END AS task_type_name,
    total_count AS same_type_work_count,
    -- 风险值计算（优先判断0分）
    CASE
        WHEN cnt_last_1year >= 2 OR cnt_last_2year >= 5 THEN 0
        WHEN total_count = 1 THEN 6
        WHEN total_count < 3 THEN 5          -- 即 total_count = 2
        WHEN total_count BETWEEN 3 AND 4 THEN 3
        WHEN total_count >= 5 THEN 2
        ELSE NULL
    END AS risk_score
FROM task_stats
ORDER BY leader_uid, task_type;

作业地段、类型，查询操作票表sp_pd_wticket_base，获取主键ID（id），查询作业计划表sp_ss_rc_work_plan，获取工作地点work_place、专业二级分类major_sub_type
、工作内容work_content，这个需要根据工作内容的描述来判断地段和类型
SQL 语句（示例）

SELECT 
    w.id AS ticket_id,
    p.work_place,
    p.major_sub_type,
    p.work_content,
    -- 根据工作内容判断地段（示例规则）
    CASE 
        WHEN p.work_content LIKE '%城区%' OR p.work_content LIKE '%市中心%' THEN '城区'
        WHEN p.work_content LIKE '%郊区%' OR p.work_content LIKE '%农村%' THEN '郊区'
        WHEN p.work_content LIKE '%山区%' THEN '山区'
        ELSE '其他地段'
    END AS work_section,
    -- 根据工作内容判断作业类型（示例规则）
    CASE 
        WHEN p.work_content LIKE '%抢修%' OR p.work_content LIKE '%故障%' THEN '抢修作业'
        WHEN p.work_content LIKE '%检修%' OR p.work_content LIKE '%维护%' THEN '检修作业'
        WHEN p.work_content LIKE '%巡视%' THEN '巡视作业'
        WHEN p.work_content LIKE '%带电作业%' THEN '带电作业'
        ELSE '常规作业'
    END AS work_type
FROM sp_pd_wticket_base w
LEFT JOIN sp_ss_rc_work_plan p ON w.ticket_source_id = p.id   -- 请根据实际外键修改
WHERE w.id = :ticket_id   -- 可选：指定工作票ID，不指定则查询所有
ORDER BY w.create_time DESC;
